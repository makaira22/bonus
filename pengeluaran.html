const NAMA_SHEET = "Pengeluaran"; const NAMA_FOLDER = "Bukti Pengeluaran"; const KOLOM_BUKTI = 7; const KOLOM_WAKTU = 8; const KOLOM_ID = 9; const KOLOM_TRANSFER = 10; // Jalankan fungsi ini SEKALI dari editor supaya izin diminta. function beriIzin() { SpreadsheetApp.getActiveSpreadsheet().getName(); try { const uji = DriveApp.createFile("uji-izin.txt", "ok"); uji.setTrashed(true); Logger.log("Drive siap. Bukti akan disimpan sebagai tautan."); } catch (err) { Logger.log("Drive ditolak: " + err); Logger.log("Tidak apa-apa. Bukti akan ditempel langsung di sheet."); } } function doPost(e) { const sheet = ambilSheet(); const isi = JSON.parse(e.postData.contents); const aksi = isi.aksi || "tambah"; if (aksi === "hapus") return balas(hapusBaris(sheet, isi.id)); if (aksi === "ubah") return balas(ubahBaris(sheet, isi.row)); return balas(tambahBaris(sheet, isi)); } function doGet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ambilSheet(); // sekalian memastikan semua kolom sudah ada const keluar = { ok: true, rows: [] }; if (sheet.getLastRow() < 2) return balas(keluar); const jumlah = sheet.getLastRow() - 1; const nilai = sheet.getRange(2, 1, jumlah, KOLOM_TRANSFER).getValues(); const rumus = sheet.getRange(2, KOLOM_BUKTI, jumlah, 1).getFormulas(); const zona = ss.getSpreadsheetTimeZone(); const daftarId = []; let adaBaru = false; for (let i = 0; i < jumlah; i++) { const b = nilai[i]; let id = String(b[KOLOM_ID - 1] || ""); // baris lama belum punya ID, diberi sekarang supaya bisa diubah dan dihapus if (b[2] && !id) { id = "sheet-" + (i + 2) + "-" + Date.now(); adaBaru = true; } daftarId.push([id]); if (!b[2]) continue; const qty = Number(b[3]) || 0; const nominal = Number(b[4]) || 0; keluar.rows.push({ id: id, tanggal: teksTanggal(b[0], zona), jenis: String(b[1] || ""), item: String(b[2]), qty: qty, nominal: nominal, bukti: urlDari(rumus[i][0]), ditransfer: nilaiTransfer(b[KOLOM_TRANSFER - 1], qty * nominal) }); } if (adaBaru) { try { sheet.getRange(2, KOLOM_ID, jumlah, 1).setValues(daftarId); } catch (err) {} } return balas(keluar); } function tambahBaris(sheet, isi) { const baris = isi.rows || [isi]; const blob = isi.bukti ? siapkanBlob(isi.bukti, isi.namaBukti) : null; const hasil = blob ? simpanKeDrive(blob) : { url: "", pesan: "" }; let barisPertama = 0; baris.forEach(function (r, i) { const qty = Number(r.qty) || 0; const nominal = Number(r.nominal) || 0; sheet.appendRow([r.tanggal, r.jenis, r.item, qty, nominal, qty * nominal, "", new Date(), r.id || ""]); const nomor = sheet.getLastRow(); if (i === 0) barisPertama = nomor; if (hasil.url) { const sel = sheet.getRange(nomor, KOLOM_BUKTI); try { sel.setFormula(tautan(hasil.url)); } catch (err) { sel.setValue(hasil.url); } } }); // Kalau Drive menolak, fotonya ditempel langsung ke sheet. // Cara ini cuma butuh izin spreadsheet yang sudah ada. let catatan = ""; if (blob && !hasil.url) { catatan = tempelGambar(sheet, blob, barisPertama, baris.length); } return { ok: true, ditambahkan: baris.length, urlBukti: hasil.url, pesanBukti: catatan ? "" : hasil.pesan, catatanBukti: catatan }; } function ubahBaris(sheet, r) { const baris = cariBaris(sheet, r.id); if (!baris) return { ok: false, pesan: "Baris tidak ditemukan di spreadsheet" }; const qty = Number(r.qty) || 0; const nominal = Number(r.nominal) || 0; sheet.getRange(baris, 1, 1, 6) .setValues([[r.tanggal, r.jenis, r.item, qty, nominal, qty * nominal]]); sheet.getRange(baris, KOLOM_WAKTU).setValue(new Date()); return { ok: true, diubah: 1, baris: baris }; } function hapusBaris(sheet, id) { const baris = cariBaris(sheet, id); if (!baris) return { ok: false, pesan: "Baris tidak ditemukan di spreadsheet" }; sheet.deleteRow(baris); return { ok: true, dihapus: 1 }; } function cariBaris(sheet, id) { if (!id || sheet.getLastRow() < 2) return 0; const nilai = sheet.getRange(2, KOLOM_ID, sheet.getLastRow() - 1, 1).getValues(); for (let i = 0; i < nilai.length; i++) { if (String(nilai[i][0]) === String(id)) return i + 2; } return 0; } function ambilSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(NAMA_SHEET) || ss.insertSheet(NAMA_SHEET); if (sheet.getMaxColumns() < KOLOM_TRANSFER) { sheet.insertColumnsAfter(sheet.getMaxColumns(), KOLOM_TRANSFER - sheet.getMaxColumns()); } if (sheet.getLastRow() === 0) { sheet.appendRow(["Tanggal","Jenis","Item","Qty","Nominal","Jumlah","Bukti","Waktu input","ID","Ditransfer"]); sheet.getRange(1, 1, 1, KOLOM_TRANSFER).setFontWeight("bold"); sheet.setFrozenRows(1); return sheet; } if (sheet.getRange(1, KOLOM_BUKTI).getValue() !== "Bukti") { sheet.insertColumnBefore(KOLOM_BUKTI); sheet.getRange(1, KOLOM_BUKTI).setValue("Bukti").setFontWeight("bold"); } if (sheet.getRange(1, KOLOM_ID).getValue() !== "ID") { sheet.getRange(1, KOLOM_ID).setValue("ID").setFontWeight("bold"); } if (sheet.getRange(1, KOLOM_TRANSFER).getValue() !== "Ditransfer") { sheet.getRange(1, KOLOM_TRANSFER).setValue("Ditransfer").setFontWeight("bold"); } return sheet; } function tempelGambar(sheet, blob, barisAwal, jumlah) { if (String(blob.getContentType()).indexOf("image/") !== 0) { sheet.getRange(barisAwal, KOLOM_BUKTI).setValue("Bukti PDF ada di aplikasi"); return ""; } try { const gambar = sheet.insertImage(blob, KOLOM_BUKTI, barisAwal); const rasio = gambar.getWidth() / gambar.getHeight(); gambar.setHeight(110); gambar.setWidth(Math.max(30, Math.round(110 * rasio))); sheet.setRowHeight(barisAwal, 118); sheet.setColumnWidth(KOLOM_BUKTI, Math.max(96, Math.round(110 * rasio) + 16)); for (let i = 1; i < jumlah; i++) { sheet.getRange(barisAwal + i, KOLOM_BUKTI).setValue("Bukti sama dengan baris di atas"); } return "Izin Drive ditolak, jadi foto ditempel langsung di sheet"; } catch (err) { sheet.getRange(barisAwal, KOLOM_BUKTI).setValue("Bukti gagal disimpan"); return ""; } } function tautan(url) { return '=HYPERLINK("' + url + '","Lihat bukti")'; } function siapkanBlob(dataUrl, nama) { try { const potong = String(dataUrl).split(","); if (potong.length < 2) return null; const tipe = (potong[0].match(/:(.*?);/) || [])[1] || "image/jpeg"; return Utilities.newBlob(Utilities.base64Decode(potong[1]), tipe, nama || ("bukti-" + Date.now() + ".jpg")); } catch (err) { return null; } } function simpanKeDrive(blob) { let file; try { file = ambilFolder().createFile(blob); } catch (err) { try { file = DriveApp.createFile(blob); } catch (err2) { return { url: "", pesan: "Gagal simpan ke Drive: " + err2 }; } } try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (err) { // file tetap ada, hanya belum bisa dibuka orang lain } return { url: file.getUrl(), pesan: "" }; } // Folder dibuat sekali lalu ID-nya diingat. Mencari folder lewat nama // butuh izin baca seluruh Drive, dan itu yang dulu bikin error. function ambilFolder() { const props = PropertiesService.getScriptProperties(); const id = props.getProperty("ID_FOLDER"); if (id) { try { return DriveApp.getFolderById(id); } catch (err) {} } const folder = DriveApp.createFolder(NAMA_FOLDER); props.setProperty("ID_FOLDER", folder.getId()); return folder; } // Kolom Ditransfer diisi finance secara manual. Boleh berupa nominal // (misal 50000 atau "Rp 50.000"), centang, atau kata seperti "sudah". function nilaiTransfer(sel, jumlah) { if (sel === true) return jumlah; if (sel === false) return 0; if (typeof sel === "number") return sel > 0 ? sel : 0; const teks = String(sel || "").trim().toLowerCase(); if (!teks) return 0; if (["ya","y","sudah","v","ok","x","lunas","done","true","transfer","cair"].indexOf(teks) !== -1) { return jumlah; } const bersih = teks.replace(/[^0-9,.]/g, "").replace(/\./g, "").replace(",", "."); const angka = Number(bersih); return isNaN(angka) || angka <= 0 ? 0 : angka; } function teksTanggal(nilai, zona) { if (nilai instanceof Date) return Utilities.formatDate(nilai, zona, "yyyy-MM-dd"); return String(nilai || "").slice(0, 10); } function urlDari(rumus) { const cocok = String(rumus || "").match(/HYPERLINK\("([^"]+)"/); return cocok ? cocok[1] : ""; } function balas(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
